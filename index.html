break; case 'editTransaction': result = editTransaction(JSON.parse(params.payload)); break; case 'deleteTransaction': result = deleteTransaction(JSON.parse(params.payload)); break; case 'updateUnbilled': result = updateUnbilled(JSON.parse(params.payload)); break; case 'settleCardBill': result = settleCardBill(params.cardRow); break; case 'toggleDailyReminder': result = toggleDailyReminder(params.enable === 'true'); break; case 'syncLiveStockPrices': result = syncLiveStockPrices(); } } function getDashboardData(queryMonth) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const dash = ss.getSheetByName('月度預算與總覽看板'); const log = ss.getSheetByName('每日收支流水帳'); const assets = ss.getSheetByName('存款與投資總資產'); const d = new Date(r[0]); if (isNaN(d.getTime())) return; const ym = r[2] ? String(r[2]).trim() : Utilities.formatDate(d, 'Asia/Taipei', 'yyyy-MM'); if (ym === targetYM) { const amt = Number(r[6]) || 0; const type = String(r[3] || ''); if (type === '支出') monthExpNum += amt; if (type === '收入') monthIncNum += amt; monthTx.push({ rowNum: idx + 3, date: Utilities.formatDate(d, 'Asia/Taipei', 'yyyy/MM/dd'), isoDate: Utilities.formatDate(d, 'Asia/Taipei', 'yyyy-MM-dd'), dayNum: d.getDate(), weekday: weekdays[d.getDay()], type: type, category: String(r[4] || ''), item: String(r[5] || ''), amount: amt, account: String(r[7] || '') }); } }); } monthTx.reverse(); // 2. 讀取三張信用卡已結帳與未出帳金額（月度預算與總覽看板 A20:G22） const cardRows = dash.getRange('A20:G22').getValues(); const cards = cardRows.map(r => ({ name: String(r[0] || ''), billed: Number(r[3]) || 0, unbilled: Number(r[6]) || 0 })); // 3. 讀取銀行與現金餘額（A10:E13） const bankRows = assets.getRange('A10:E13').getValues(); const banks = []; let cashTotal = 0; bankRows.forEach(r => { const name = String(r[0] || '').trim(); if (name && name !== '0' && name !== '帳戶名稱' && !name.includes('小計')) { const bal = Number(r[4]) || 0; cashTotal += bal; banks.push({ name: name, balance: bal.toLocaleString() }); } }); // 4. 讀取台美股 ETF 投資組合（A20:H29） const invRows = assets.getRange('A20:H29').getValues(); const investments = []; let investTotal = 0; invRows.forEach(r => { const rawCode = String(r[0] || '').trim(); if (rawCode && rawCode !== '0' && rawCode !== '股票代號' && !rawCode.includes('小計')) { const mVal = Math.round(Number(r[6]) || 0); investTotal += mVal; investments.push({ name: String(r[1] || ''), ticker: rawCode.replace(/^TPE:/i, ''), price: Number(r[5] || 0).toFixed(2), marketVal: mVal.toLocaleString(), pnl: Math.round(Number(r[7]) || 0) }); } }); const sheetNetWorth = Number(assets.getRange('E5').getValue()); const netWorthVal = !isNaN(sheetNetWorth) ? sheetNetWorth : (cashTotal + investTotal); const expenseCats = ['飲食', '交通', '居家生活', '娛樂購物', '醫療保險', '學習進修', '其他支出']; const incomeCats = ['薪資', '獎金', '股息利息', '副業收入', '其他收入']; const accounts = ['現金', '台新銀行', '國泰世華', '中國信託', '台新玫瑰 Giving 卡', '國泰 CUBE 卡', '中國信託 LINE Pay 卡']; const hasReminder = ScriptApp.getProjectTriggers().some(t => t.getHandlerFunction() === 'sendDailyReminder'); return { queryMonth: targetYM, syncTime: Utilities.formatDate(now, 'Asia/Taipei', 'HH:mm:ss'), monthExpNum: monthExpNum, monthIncNum: monthIncNum, monthTx: monthTx, netWorth: netWorthVal.toLocaleString(), cashTotal: cashTotal.toLocaleString(), investTotal: investTotal.toLocaleString(), cards: cards, banks: banks, investments: investments, expenseCats: expenseCats, incomeCats: incomeCats, accounts: accounts, hasReminder: hasReminder }; } // 輔助函式：安全調整信用卡未出帳金額（delta 為正代表增加，為負代表扣回） function adjustCardUnbilled(dash, type, account, deltaAmt) { if (type !== '支出' || !deltaAmt) return; const cardRowMap = { '台新玫瑰 Giving 卡': 20, '國泰 CUBE 卡': 21, '中國信託 LINE Pay 卡': 22 }; const cardRow = cardRowMap[account]; if (cardRow) { const cell = dash.getRange(cardRow, 7); const current = Number(cell.getValue()) || 0; cell.setValue(Math.max(0, current + deltaAmt)); } } function addTransaction(tx) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const log = ss.getSheetByName('每日收支流水帳'); const dash = ss.getSheetByName('月度預算與總覽看板'); const dates = log.getRange('A3:A252').getValues(); let targetRow = -1; for (let i = 0; i < dates.length; i++) { if (dates[i][0] === '' || dates[i][0] === null) { targetRow = i + 3; break; } } if (targetRow === -1) throw new Error('每日收支流水帳已滿，請先擴充列數！'); const amt = Number(tx.amount) || 0; log.getRange(targetRow, 1).setValue(new Date(tx.date)); log.getRange('B6:C6').copyTo(log.getRange(targetRow, 2, 1, 2)); log.getRange(targetRow, 4, 1, 6).setValues([[tx.type, tx.category, tx.item, amt, tx.account, '手機App記帳']]); adjustCardUnbilled(dash, tx.type, tx.account, amt); SpreadsheetApp.flush(); return getDashboardData(tx.date.slice(0, 7)); } function editTransaction(tx) { const row = Number(tx.rowNum); if (!row || row < 3) throw new Error('無效的明細列號'); const ss = SpreadsheetApp.getActiveSpreadsheet(); const log = ss.getSheetByName('每日收支流水帳'); const dash = ss.getSheetByName('月度預算與總覽看板'); // 讀取該列舊資料，若原本是信用卡支出則先扣回舊金額 const oldVals = log.getRange(row, 1, 1, 9).getValues()[0]; if (!oldVals[0]) throw new Error('找不到該筆紀錄，可能已被刪除'); const oldType = String(oldVals[3] || ''); return getDashboardData(tx.queryMonth || tx.date.slice(0, 7)); } function deleteTransaction(payload) { const row = Number(payload.rowNum); if (!row || row < 3) throw new Error('無效的明細列號'); const ss = SpreadsheetApp.getActiveSpreadsheet(); log.getRange(row, 4, 1, 6).clearContent(); SpreadsheetApp.flush(); return getDashboardData(payload.queryMonth || ''); } function syncLiveStockPrices() { const ss = SpreadsheetApp.getActiveSpreadsheet();
